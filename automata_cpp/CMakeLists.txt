cmake_minimum_required(VERSION 3.16)
project(automata_cpp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
foreach(program regex_to_nfa nfa_to_dfa dfa_minimize automaton_to_regex regex_to_min_dfa dfa_to_nfa)
    add_executable(${program} ${program}.cpp)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${program} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()
include(CTest)
if(BUILD_TESTING)
    add_executable(test_automata tests/test_automata.cpp)
    add_test(NAME automata_properties COMMAND test_automata)
endif()
